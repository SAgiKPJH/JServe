```bash
# 프로젝트 클론

cd jserve

# 의존성 설치

# 환경 변수 설정
```

### React 컴포넌트 구조

```tsx
// src/client/components/CodeEditor/CodeEditor.tsx
import React, { useState, useCallback } from 'react';
import { Editor } from '@monaco-editor/react';
import { useCodeExecution } from '../../hooks/useCodeExecution';

interface CodeEditorProps {
  language: string;
  defaultValue?: string;
  onCodeChange?: (code: string) => void;
}

export const CodeEditor: React.FC<CodeEditorProps> = ({
  language,
  defaultValue = '',
  onCodeChange
}) => {
  const [code, setCode] = useState(defaultValue);
  const { executeCode, isExecuting, result } = useCodeExecution();

  const handleCodeChange = useCallback((value: string | undefined) => {
    const newCode = value || '';
    setCode(newCode);
    onCodeChange?.(newCode);
  }, [onCodeChange]);

  const handleExecute = useCallback(() => {
    executeCode(language, code);
  }, [language, code, executeCode]);

  return (
    <div className="code-editor">
      <Editor
        height="400px"
        language={language}
        value={code}
        onChange={handleCodeChange}
        theme="vs-dark"
        options={{
          minimap: { enabled: false },
          fontSize: 14,
          wordWrap: 'on'
        }}
      />
      <button
        onClick={handleExecute}
        disabled={isExecuting}
        className="execute-button"
      >
        {isExecuting ? 'Running...' : 'Run Code'}
      </button>
      {result && (
        <div className="execution-result">
          {result.output && <pre>{result.output}</pre>}
          {result.error && <pre className="error">{result.error}</pre>}
        </div>
      )}
    </div>
  );
};
```

### 상태 관리 (Zustand)

```tsx
// src/client/store/codeStore.ts
import { create } from 'zustand';

interface CodeState {
  currentLanguage: string;
  code: string;
  files: CodeFile[];
  executionHistory: ExecutionResult[];
  setLanguage: (language: string) => void;
  setCode: (code: string) => void;
  addFile: (file: CodeFile) => void;
  removeFile: (id: string) => void;
  addExecutionResult: (result: ExecutionResult) => void;
}

export const useCodeStore = create<CodeState>((set) => ({
  currentLanguage: 'javascript',
  code: '',
  files: [],
  executionHistory: [],

  setLanguage: (language) => set({ currentLanguage: language }),
  setCode: (code) => set({ code }),

  addFile: (file) => set((state) => ({
    files: [...state.files, file]
  })),

  removeFile: (id) => set((state) => ({
    files: state.files.filter(f => f.id !== id)
  })),

  addExecutionResult: (result) => set((state) => ({
    executionHistory: [result, ...state.executionHistory].slice(0, 50)
  }))
}));
```

```javascript
// Routes
app.use('/api/auth', authRoutes);
app.use('/api/code', authMiddleware, codeRoutes);
app.use('/api/files', authMiddleware, fileRoutes);

// Error handling
```

### 코드 실행 서비스

```javascript
// src/server/services/codeExecutionService.js
const Docker = require('dockerode');
const { v4: uuidv4 } = require('uuid');

class CodeExecutionService {
  constructor() {
    this.docker = new Docker();
    this.runnerImages = {
      javascript: 'jserve-runner:node',
      python: 'jserve-runner:python',
      java: 'jserve-runner:java'
    };
  }

  async executeCode(language, code, input = '') {
    const executionId = uuidv4();
    const imageName = this.runnerImages[language];

    if (!imageName) {
      throw new Error(`Unsupported language: ${language}`);
    }

    const container = await this.docker.createContainer({
      Image: imageName,
      Cmd: ['/bin/sh', '-c', this.getExecutionCommand(language, code)],
      Env: [`INPUT=${input}`],
      WorkingDir: '/app',
      HostConfig: {
        Memory: 128 * 1024 * 1024, // 128MB
        CpuQuota: 50000, // 50% CPU
        NetworkMode: 'none', // 네트워크 접근 차단
        ReadonlyRootfs: true,
        Tmpfs: { '/tmp': 'rw,noexec,nosuid,size=10m' }
      },
      AttachStdout: true,
      AttachStderr: true
    });

    try {
      await container.start();

      const stream = await container.attach({
        stream: true,
        stdout: true,
        stderr: true
      });

      const output = await this.collectOutput(stream, 10000); // 10초 타임아웃

      const inspection = await container.inspect();
      const exitCode = inspection.State.ExitCode;

      await container.remove();

      return {
        executionId,
        success: exitCode === 0,
        output: output.stdout,
        error: output.stderr,
        executionTime: this.calculateExecutionTime(inspection),
        language
      };

    } catch (error) {
      await container.remove();
      throw error;
    }
  }

  getExecutionCommand(language, code) {
    const commands = {
      javascript: `echo '${this.escapeCode(code)}' > main.js && node main.js`,
      python: `echo '${this.escapeCode(code)}' > main.py && python main.py`,
      java: `echo '${this.escapeCode(code)}' > Main.java && javac Main.java && java Main`
    };

    return commands[language];
  }

  escapeCode(code) {
    return code.replace(/'/g, "'\"'\"'");
  }

  async collectOutput(stream, timeout = 10000) {
    return new Promise((resolve, reject) => {
      let stdout = '';
      let stderr = '';

      const timer = setTimeout(() => {
        reject(new Error('Execution timeout'));
      }, timeout);

      stream.on('data', (chunk) => {
        const data = chunk.toString();
        if (chunk[0] === 1) {
          stdout += data.slice(8);
        } else if (chunk[0] === 2) {
          stderr += data.slice(8);
        }
      });

      stream.on('end', () => {
        clearTimeout(timer);
        resolve({ stdout, stderr });
      });

      stream.on('error', (error) => {
        clearTimeout(timer);
        reject(error);
      });
    });
  }

  calculateExecutionTime(inspection) {
    const startTime = new Date(inspection.State.StartedAt);
    const finishTime = new Date(inspection.State.FinishedAt);
    return finishTime - startTime;
  }
}

module.exports = new CodeExecutionService();
```

## 🧪 테스트 전략

### 단위 테스트 (Jest)

```javascript
// tests/unit/codeExecutionService.test.js
const CodeExecutionService = require('../../src/server/services/codeExecutionService');

describe('CodeExecutionService', () => {
  describe('executeCode', () => {
    it('should execute JavaScript code successfully', async () => {
      const code = 'console.log("Hello, World!");';
      const result = await CodeExecutionService.executeCode('javascript', code);

      expect(result.success).toBe(true);
      expect(result.output).toContain('Hello, World!');
      expect(result.language).toBe('javascript');
    });

    it('should handle syntax errors', async () => {
      const code = 'console.log("Hello, World!";'; // 구문 오류
      const result = await CodeExecutionService.executeCode('javascript', code);

      expect(result.success).toBe(false);
      expect(result.error).toContain('SyntaxError');
    });

    it('should enforce timeout limits', async () => {
      const code = 'while(true) {}'; // 무한 루프

      await expect(
        CodeExecutionService.executeCode('javascript', code)
      ).rejects.toThrow('Execution timeout');
    });
  });
});
```
